int stareLed = 0; 
const int ledPin = 13;
void setup() {
  Serial.begin(9600);
  pinMode(ledPin, OUTPUT);
}
void loop() {
  if (Serial.available() > 0) {
    char optiune = Serial.read();
    
    if (optiune == '1') {
      Serial.println("pornit");
      stareLed = 1;
    } 
    else if (optiune == '2') {
      Serial.println("oprit");
      stareLed = 2;
    } 
    else if (optiune == '3') {
      Serial.println("blink");
      stareLed = 3;
    }
  }
 
  if (stareLed == 1) {
    digitalWrite(ledPin, HIGH); 
  } 
  else if (stareLed == 2) {
    digitalWrite(ledPin, LOW);
  } 
  else if (stareLed == 3) {
    digitalWrite(ledPin, HIGH);
    delay(10000);          
    digitalWrite(ledPin, LOW);
    delay(10000);
  }
}
